file_input_rds <- file.path(dir_data_clean, "ms4_imported_clean.rds")

required_qc_columns <- c(
  "row_id",
  "mobile_cell_count",
  "stationary_cell_count",
  "total_cells",
  "frame_count",
  "count_date_yyyy_mm_dd",
  "start_date_yyyy_mm_dd",
  "experiment_type",
  "days_from_start",
  "lux_exposure",

  "particle_type",

  "mass_loading_ug_l"
)

missing_qc_columns <- required_qc_columns[!required_qc_columns %in% names(dat)]

if (length(missing_qc_columns) > 0) {

# ---------------------------------------------------------
# 6. Standardise selected fields for QC evaluation
# ---------------------------------------------------------

standardise_upper_trim <- function(x) {
  x <- as.character(x)
  x <- stringr::str_squish(x)
  x <- toupper(x)
  x[x %in% c("", "NA", "N/A", "NULL", "null", ".")] <- NA_character_
  x
}

for (nm in intersect(c("experiment_type", "particle_type", "size_class", "toxin_exposure"), names(dat))) {
  dat[[nm]] <- standardise_upper_trim(dat[[nm]])
}

cat("Selected fields standardised for QC checks.\n\n")

# ---------------------------------------------------------
# 7. Refresh count-derived QC check values
# ---------------------------------------------------------

dat <- dat |>
  dplyr::mutate(
    qc_recalc_total_cells = dplyr::if_else(
      !is.na(mobile_cell_count) & !is.na(stationary_cell_count),
      mobile_cell_count + stationary_cell_count,
      NA_integer_
    ),
    qc_recalc_motility_ratio = dplyr::if_else(
      !is.na(mobile_cell_count) & !is.na(stationary_cell_count) &
        (mobile_cell_count + stationary_cell_count) > 0,
      mobile_cell_count / (mobile_cell_count + stationary_cell_count),
      NA_real_
    )
  )

cat("QC recalculated totals created.\n\n")

# ---------------------------------------------------------
# 8. Expected frame count rule
# ---------------------------------------------------------

expected_frame_count <- 500L
frame_tolerance <- 10L

cat("Frame count QC rule:\n")
cat("- Expected frame count:", expected_frame_count, "\n")
cat("- Tolerance:", frame_tolerance, "frames\n\n")

# ---------------------------------------------------------
# 9. Create pre-specified QC flags
# ---------------------------------------------------------

dat_qc <- dat |>
  dplyr::mutate(
    flag_missing_response_counts =
      is.na(mobile_cell_count) | is.na(stationary_cell_count),

    flag_total_cell_count_le_zero =
      !is.na(qc_recalc_total_cells) & qc_recalc_total_cells <= 0,

    flag_inconsistent_count_totals =
      !is.na(total_cells) & !is.na(qc_recalc_total_cells) &
      total_cells != qc_recalc_total_cells,

    flag_missing_key_predictor =
      dplyr::case_when(
        experiment_type %in% c("LIGHT_ONLY", "EXP1", "LIGHT") ~
          is.na(lux_exposure) | is.na(days_from_start),

        experiment_type %in% c("DEFINED_PARTICLES", "EXP2") ~
          is.na(ntu) | is.na(particle_type) | is.na(days_from_start),

        experiment_type %in% c("BRAKE_SIZE", "EXP3") ~
          is.na(size_class) | is.na(days_from_start),

        experiment_type %in% c("FIELD_SPM", "EXP4") ~
          is.na(ntu) | is.na(days_from_start),

        TRUE ~ TRUE
      ),

    flag_unexpected_frame_count =
      is.na(frame_count) |
      abs(frame_count - expected_frame_count) > frame_tolerance,

    flag_impossible_date_sequences =
      !is.na(count_date_yyyy_mm_dd) &
      !is.na(start_date_yyyy_mm_dd) &
      count_date_yyyy_mm_dd < start_date_yyyy_mm_dd,

    flag_negative_mobile_count =
      !is.na(mobile_cell_count) & mobile_cell_count < 0,

    flag_negative_stationary_count =
      !is.na(stationary_cell_count) & stationary_cell_count < 0,

    flag_negative_ntu =
      !is.na(ntu) & ntu < 0,

    flag_negative_lux =
      !is.na(lux_exposure) & lux_exposure < 0,

    flag_negative_mass_loading =
      !is.na(mass_loading_ug_l) & mass_loading_ug_l < 0
  )

cat("Pre-specified QC flags created.\n\n")

# ---------------------------------------------------------
# 10. Create row-level flag counts and flag reason field
# ---------------------------------------------------------

flag_columns <- names(dat_qc)[stringr::str_detect(names(dat_qc), "^flag_")]

dat_qc <- dat_qc |>
  dplyr::mutate(
    qc_n_flags = rowSums(dplyr::across(dplyr::all_of(flag_columns), ~ as.integer(.x)), na.rm = TRUE),
    qc_any_flag = qc_n_flags > 0
  )

collapse_true_reasons <- function(row_values, prefix = "^flag_") {
  nm <- names(row_values)[which(row_values)]
  if (length(nm) == 0) {
    return(NA_character_)
  }
  nm <- gsub(prefix, "", nm)
  paste(nm, collapse = "; ")
}

flag_matrix <- as.data.frame(dat_qc[, flag_columns, drop = FALSE])

dat_qc$qc_flag_reason <- vapply(
  seq_len(nrow(flag_matrix)),
  function(i) collapse_true_reasons(as.logical(unlist(flag_matrix[i, ])), prefix = "^flag_"),
  character(1)
)

dat_qc$qc_flag_reason[!dat_qc$qc_any_flag] <- NA_character_

if (exists("prettify_reason_string", mode = "function", inherits = TRUE)) {
  dat_qc$qc_flag_reason_pretty <- prettify_reason_string(dat_qc$qc_flag_reason)
} else {
  dat_qc$qc_flag_reason_pretty <- dat_qc$qc_flag_reason
}

cat("QC flag counts and reason fields created.\n\n")

# ---------------------------------------------------------
# 11. Define explicit exclusion rules
# ---------------------------------------------------------

dat_qc <- dat_qc |>
  dplyr::mutate(
    exclude_missing_response_counts = flag_missing_response_counts,
    exclude_total_cell_count_le_zero = flag_total_cell_count_le_zero,
    exclude_missing_key_predictor = flag_missing_key_predictor,
    exclude_negative_mobile_count = flag_negative_mobile_count,
    exclude_negative_stationary_count = flag_negative_stationary_count,

    # flagged but not excluded by default
    exclude_inconsistent_count_totals = FALSE,
    exclude_unexpected_frame_count = FALSE,
    exclude_impossible_date_sequences = FALSE,
    exclude_negative_ntu = FALSE,
    exclude_negative_lux = FALSE,
    exclude_negative_mass_loading = FALSE
  )

exclude_columns <- names(dat_qc)[stringr::str_detect(names(dat_qc), "^exclude_")]

dat_qc <- dat_qc |>
  dplyr::mutate(
    exclusion_flag = rowSums(dplyr::across(dplyr::all_of(exclude_columns), ~ as.integer(.x)), na.rm = TRUE) > 0
  )

exclude_matrix <- as.data.frame(dat_qc[, exclude_columns, drop = FALSE])

dat_qc$exclusion_reason <- vapply(
  seq_len(nrow(exclude_matrix)),
  function(i) collapse_true_reasons(as.logical(unlist(exclude_matrix[i, ])), prefix = "^exclude_"),
  character(1)
)

dat_qc$exclusion_reason[!dat_qc$exclusion_flag] <- NA_character_

if (exists("prettify_reason_string", mode = "function", inherits = TRUE)) {
  dat_qc$exclusion_reason_pretty <- prettify_reason_string(dat_qc$exclusion_reason)
} else {
  dat_qc$exclusion_reason_pretty <- dat_qc$exclusion_reason
}

cat("Explicit exclusion rules applied.\n\n")

# ---------------------------------------------------------
# 12. Create row-level QC table
# ---------------------------------------------------------

row_level_flags <- dat_qc |>
  dplyr::select(
    row_id,
    dplyr::any_of(c(
      "video_file",
      "experiment_id",
      "experiment_type",
      "culture",
      "well",
      "days_from_start"

    dplyr::all_of(flag_columns),
    qc_n_flags,
    qc_any_flag,
    qc_flag_reason,
    qc_flag_reason_pretty,
    dplyr::all_of(exclude_columns),
    exclusion_flag,
    exclusion_reason,

readr::write_csv(row_level_flags, file_qc_row_level_flags)

cat("Row-level QC flags written to:\n")
cat(file_qc_row_level_flags, "\n\n")

# ---------------------------------------------------------
# 13. Build QC summary tables
# ---------------------------------------------------------

qc_flag_summary <- tibble::tibble(
  qc_flag = flag_columns,
  n_flagged = vapply(flag_columns, function(x) sum(dat_qc[[x]], na.rm = TRUE), integer(1)),
  pct_flagged = round(
    vapply(flag_columns, function(x) mean(dat_qc[[x]], na.rm = TRUE), numeric(1)) * 100,
    2
  )
) |>
  dplyr::mutate(
    qc_flag_label = if (exists("label_qc_flag", mode = "function", inherits = TRUE)) {
      label_qc_flag(qc_flag)
    } else {
      qc_flag
    }
  ) |>
  dplyr::arrange(dplyr::desc(n_flagged), qc_flag)

exclusion_summary <- tibble::tibble(
  exclusion_rule = exclude_columns,
  n_excluded_by_rule = vapply(exclude_columns, function(x) sum(dat_qc[[x]], na.rm = TRUE), integer(1)),
  pct_excluded_by_rule = round(
    vapply(exclude_columns, function(x) mean(dat_qc[[x]], na.rm = TRUE), numeric(1)) * 100,
    2
  )
) |>
  dplyr::mutate(
    exclusion_label = if (exists("label_exclusion", mode = "function", inherits = TRUE)) {
      label_exclusion(exclusion_rule)
    } else {
      exclusion_rule
    }
  ) |>
  dplyr::arrange(dplyr::desc(n_excluded_by_rule), exclusion_rule)

readr::write_csv(qc_flag_summary, file_qc_flag_summary)
readr::write_csv(exclusion_summary, file_exclusion_summary)
