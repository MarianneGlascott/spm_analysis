save_main_figure <- function(plot_obj, figure_name, caption_text) {
  if (is.null(plot_obj)) return(character(0))

  saved <- character(0)

  if (exists("save_figure_both_widths", mode = "function", inherits = TRUE)) {
    saved <- save_figure_both_widths(
      plot = plot_obj,
      figure_name = figure_name,
      subdir = figure_subdir,
      height = "standard",
      quiet = TRUE
    )
  }

  if (exists("write_figure_caption_md", mode = "function", inherits = TRUE)) {
    write_figure_caption_md(
      figure_name = figure_name,
      caption_text = caption_text,
      subdir = figure_subdir
    )
  }

  saved
}

)
