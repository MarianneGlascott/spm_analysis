read_if_exists <- function(path) {
  if (!file.exists(path)) return(NULL)
  readr::read_csv(path, show_col_types = FALSE, progress = FALSE)
}

collapse_sorted_unique <- function(x, sep = ", ") {
  x <- unique(x[!is.na(x)])
  x <- x[order(x)]
  if (length(x) == 0) return(NA_character_)
  paste(x, collapse = sep)
}

safe_n_distinct <- function(x) {
  dplyr::n_distinct(x, na.rm = TRUE)
}

safe_min <- function(x) {
  x <- x[!is.na(x)]
  if (length(x) == 0) return(NA_real_)
  min(x)
}

safe_max <- function(x) {
  x <- x[!is.na(x)]
  if (length(x) == 0) return(NA_real_)
  max(x)
}

build_dataset_summary <- function(df, experiment_label) {
  if (nrow(df) == 0) return(NULL)

  df |>
    dplyr::group_by(days_from_start) |>
    dplyr::summarise(
      experiment_label = experiment_label,
      n_cultures = safe_n_distinct(culture),
      n_wells = if ("well" %in% names(.)) safe_n_distinct(well) else NA_integer_,
      n_videos = if ("video_file" %in% names(.)) safe_n_distinct(video_file) else NA_integer_,
      n_observations = dplyr::n(),
      .groups = "drop"
    ) |>
    dplyr::select(
      experiment_label,
      days_from_start,
      n_cultures,
      n_wells,
      n_videos,
      n_observations
    )
}

table2 <- dplyr::bind_rows(

standardise_model_summary <- function(df, experiment_label, experiment_num_value, model_family = "binomial (logit)", random_effects = "(1 | culture)") {
  if (is.null(df) || nrow(df) == 0) return(NULL)

  out <- df |>
    dplyr::mutate(
      experiment_label = experiment_label,
      experiment_num = experiment_num_value,
      model_family = model_family,
      random_effects = random_effects
    )

  nm <- names(out)

  if (!"term" %in% nm) {
    if ("parameter" %in% nm) out <- dplyr::rename(out, term = parameter)
    if ("effect" %in% nm) out <- dplyr::rename(out, term = effect)
  }
  if (!"estimate" %in% names(out)) {
    if ("estimate_response" %in% names(out)) out <- dplyr::rename(out, estimate = estimate_response)
  }
  if (!"conf_low" %in% names(out)) {
    if ("conf.low" %in% names(out)) out <- dplyr::rename(out, conf_low = conf.low)
    if ("conf_low_response" %in% names(out)) out <- dplyr::rename(out, conf_low = conf_low_response)
  }
  if (!"conf_high" %in% names(out)) {
    if ("conf.high" %in% names(out)) out <- dplyr::rename(out, conf_high = conf.high)
    if ("conf_high_response" %in% names(out)) out <- dplyr::rename(out, conf_high = conf_high_response)
  }
  if (!"statistic" %in% names(out)) {
    if ("z.value" %in% names(out)) out <- dplyr::rename(out, statistic = z.value)
    if ("test_statistic" %in% names(out)) out <- dplyr::rename(out, statistic = test_statistic)
  }
  if (!"p_value" %in% names(out)) {
    if ("p.value" %in% names(out)) out <- dplyr::rename(out, p_value = p.value)
    if ("p_value_raw" %in% names(out)) out <- dplyr::rename(out, p_value = p_value_raw)
  }

  keep_cols <- intersect(
    c(
      "experiment_label",
      "experiment_num",
      "model_name",
      "term",
      "estimate",
      "conf_low",
      "conf_high",
      "statistic",
      "p_value",
      "model_family",
      "random_effects",
      "note"
    ),
    names(out)
  )

  out |>
    dplyr::select(dplyr::all_of(keep_cols))
}

# 12. Table 4: Model comparison / sensitivity summary
# ---------------------------------------------------------

standardise_model_comparison <- function(df, experiment_label, experiment_num_value) {
  if (is.null(df) || nrow(df) == 0) return(NULL)

  out <- df |>
    dplyr::mutate(
      experiment_label = experiment_label,
      experiment_num = experiment_num_value
    )

  if (!"model_rank" %in% names(out)) {
    out <- out |>
      dplyr::mutate(model_rank = dplyr::row_number())
  }

  keep_cols <- intersect(
    c(
      "experiment_label",
      "experiment_num",
      "model_rank",
      "model",
      "aic",
      "delta_aic",
      "weight",
      "logLik",
      "deviance",
      "df",
      "npar"
    ),
    names(out)
  )

  out |>
    dplyr::select(dplyr::all_of(keep_cols))
}

  standardise_model_comparison(modcomp_exp3, "Experiment 3: Brake-wear size", 11.2),
