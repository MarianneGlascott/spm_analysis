# ---------------------------------------------------------
# 7. Standardise size class and derive support vars
# ---------------------------------------------------------

standardise_upper_trim <- function(x) {
  x <- as.character(x)
  x <- stringr::str_squish(x)
  x <- toupper(x)
  x[x %in% c("", "NA", "N/A", "NULL", "null", ".")] <- NA_character_
  x
}

exp3_dat <- exp3_dat |>
  dplyr::mutate(
    particle_size_class = standardise_upper_trim(size_class),
    particle_size_class = dplyr::case_when(
      particle_size_class %in% c("COARSE", "BWC", "BRAKE WEAR COARSE", "BRAKE_WEAR_COARSE") ~ "BWC",
      particle_size_class %in% c("FINE", "BWF", "BRAKE WEAR FINE", "BRAKE_WEAR_FINE") ~ "BWF",
      TRUE ~ NA_character_
    ),
    total_cells = mobile_cell_count + stationary_cell_count,
    motility_ratio = dplyr::if_else(
      total_cells > 0,
      mobile_cell_count / total_cells,
      NA_real_
    ),
    days_from_start_f = factor(
      days_from_start,
      levels = sort(unique(days_from_start))
    ),
    days_from_start_c = as.numeric(scale(days_from_start, scale = FALSE)),
    day_label = paste0("Day ", days_from_start),
    experiment_plot_label = "Experiment 3: Brake size"
  ) |>
  dplyr::filter(
    !is.na(particle_size_class),
    particle_size_class %in% c("BWC", "BWF")
  ) |>
  dplyr::mutate(
    particle_size_class = factor(
      particle_size_class,
      levels = c("BWC", "BWF")
    )
  )

observed_size_levels <- levels(droplevels(exp3_dat$particle_size_class))

}

# ---------------------------------------------------------
# 9. Define candidate model formulas
# ---------------------------------------------------------

formula_full_primary <- cbind(mobile_cell_count, stationary_cell_count) ~
  particle_size_class * log10_bw_ug_l_plus1 * days_from_start_c +
  (1 | culture)

formula_additive_primary <- cbind(mobile_cell_count, stationary_cell_count) ~
  particle_size_class + log10_bw_ug_l_plus1 + days_from_start_c +
  (1 | culture)

formula_null <- cbind(mobile_cell_count, stationary_cell_count) ~
  1 + (1 | culture)

cat("Candidate model formulas defined.\n\n")
