read_any_data <- function(rds_path, csv_path, object_label) {
  if (file.exists(rds_path)) {
    x <- readRDS(rds_path)
    attr(x, "source_path") <- rds_path
    return(x)
  }
  if (file.exists(csv_path)) {
    x <- readr::read_csv(csv_path, show_col_types = FALSE, progress = FALSE)
    attr(x, "source_path") <- csv_path
    return(x)
  }
  stop(
    paste0(
      "Required file for ", object_label, " not found.\nExpected one of:\n- ",
      rds_path, "\n- ", csv_path
    ),
    call. = FALSE
  )
}

read_if_exists <- function(path) {
  if (!file.exists(path)) return(NULL)
  readr::read_csv(path, show_col_types = FALSE, progress = FALSE)
}

safe_n_distinct <- function(x) {
  dplyr::n_distinct(x, na.rm = TRUE)
}

safe_mean <- function(x) {
  if (all(is.na(x))) return(NA_real_)
  mean(x, na.rm = TRUE)
}

safe_sd <- function(x) {
  if (sum(!is.na(x)) < 2) return(NA_real_)
  stats::sd(x, na.rm = TRUE)
}

safe_min <- function(x) {
  x <- x[!is.na(x)]
  if (length(x) == 0) return(NA_real_)
  min(x)
}

safe_max <- function(x) {
  x <- x[!is.na(x)]
  if (length(x) == 0) return(NA_real_)
  max(x)
}

save_supp_figure <- function(plot_obj, figure_name, caption_text, height = "standard") {
  if (is.null(plot_obj)) return(character(0))

  saved <- character(0)

  if (exists("save_figure_both_widths", mode = "function", inherits = TRUE)) {
    saved <- save_figure_both_widths(
      plot = plot_obj,
      figure_name = figure_name,
      subdir = figure_subdir,
      height = height,
      quiet = TRUE
    )
  }

  if (exists("write_figure_caption_md", mode = "function", inherits = TRUE)) {
    write_figure_caption_md(
      figure_name = figure_name,
      caption_text = caption_text,
      subdir = figure_subdir
    )
  }

  saved
}

standardise_model_comp <- function(df, experiment_label, experiment_num_value) {
  if (is.null(df) || nrow(df) == 0) return(NULL)

  out <- df |>
    dplyr::mutate(
      experiment_label = experiment_label,
      experiment_num = experiment_num_value
    )

  if (!"model_rank" %in% names(out)) {
    out <- out |>
      dplyr::mutate(model_rank = dplyr::row_number())
  }

  out
}

table_s3 <- dplyr::bind_rows(
  standardise_model_comp(modcomp_exp1, "Experiment 1: Light-only gradient", 9.2),
  standardise_model_comp(modcomp_exp2, "Experiment 2: Defined particles", 10.2),
  standardise_model_comp(modcomp_exp3, "Experiment 3: Brake-wear size", 11.2),
  standardise_model_comp(modcomp_exp4, "Experiment 4: Field-derived SPM", 8.2)
) |>
  dplyr::arrange(experiment_num, model_rank)
